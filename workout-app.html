Google Sheets Backup Setup

One-time setup (5 min):

1. Open Google Sheets and create a new blank spreadsheet. Name it whatever you want (e.g. "Workout Backup").

2. Go to Extensions > Apps Script.

3. Delete everything in the editor and paste this code:

function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.action === 'push') { // --- Readable sheet: one row per exercise --- var vis = ss.getSheetByName('Workouts'); if (!vis) vis = ss.insertSheet('Workouts'); vis.clear(); vis.appendRow(['Date','Day','Duration (min)', 'Exercise','Type','Weight (lbs)', 'Set 1','Set 2','Set 3','Set 4','Notes']); // Bold + freeze header vis.getRange(1,1,1,11).setFontWeight('bold'); vis.setFrozenRows(1); var rows = []; data.workouts.forEach(function(w) { w.exercises.forEach(function(ex) { var r = [w.date, w.dayLabel||'', w.duration||'', ex.name, ex.type||'', ex.weight||'']; var sets = ex.sets || []; for (var i = 0; i < 4; i++) { if (!sets[i]) { r.push(''); continue; } var hasSetWeight = sets[i].weight !== undefined && sets[i].weight !== ex.weight; r.push(hasSetWeight ? sets[i].reps + '@' + sets[i].weight : sets[i].reps); } r.push(ex.note || ''); rows.push(r); }); }); if (rows.length > 0) vis.getRange(2,1,rows.length,11).setValues(rows); // --- Raw sheet: JSON for reliable restore --- var raw = ss.getSheetByName('_raw'); if (!raw) { raw = ss.insertSheet('_raw'); } raw.clear(); var json = JSON.stringify(data.workouts); var CHUNK = 40000; for (var i = 0; i < json.length; i += CHUNK) raw.getRange(i/CHUNK+1,1).setValue(json.substring(i,i+CHUNK)); raw.getRange(1,2).setValue(new Date().toISOString()); raw.getRange(1,3).setValue(Math.ceil(json.length/CHUNK)); return ContentService.createTextOutput(JSON.stringify({ ok:true, count:data.workouts.length, time:new Date().toISOString() })).setMimeType(ContentService.MimeType.JSON); } if (data.action === 'pull') { var raw = ss.getSheetByName('_raw'); if (!raw) return ContentService.createTextOutput( JSON.stringify({ok:true, workouts:[], time:null}) ).setMimeType(ContentService.MimeType.JSON); var n = parseInt(raw.getRange(1,3).getValue()) || 1; var json = ''; for (var i = 0; i < n; i++) json += (raw.getRange(i+1,1).getValue() || ''); var time = raw.getRange(1,2).getValue(); return ContentService.createTextOutput(JSON.stringify({ ok:true, workouts: json ? JSON.parse(json) : [], time: time||null })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput( JSON.stringify({ok:false}) ).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var cb = (e.parameter || {}).callback || ''; var action = (e.parameter || {}).action || 'ping'; var result; if (action === 'pull') { var ss = SpreadsheetApp.getActiveSpreadsheet(); var raw = ss.getSheetByName('_raw'); if (!raw) { result = {ok:true, workouts:[], time:null}; } else { var n = parseInt(raw.getRange(1,3).getValue()) || 1; var json = ''; for (var i = 0; i < n; i++) json += (raw.getRange(i+1,1).getValue() || ''); var time = raw.getRange(1,2).getValue(); result = {ok:true, workouts: json ? JSON.parse(json) : [], time: time||null}; } } else { result = {ok:true, msg:'IronLog Sync active'}; } // JSONP: wrap in callback if provided if (cb) { return ContentService.createTextOutput( cb + '(' + JSON.stringify(result) + ')' ).setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput( JSON.stringify(result) ).setMimeType(ContentService.MimeType.JSON); }

4. Click the 💾 Save icon (or Ctrl+S), then click Deploy > New deployment.

5. Click the ⚙️ gear next to "Select type" and choose Web app.

6. Set "Execute as" to Me. Set "Who has access" to Anyone. Click Deploy.

7. Click Authorize access when prompted. If you see "Google hasn't verified this app", click Advanced > Go to Untitled project (unsafe) — this is safe, it's your own script.

8. Copy the Web app URL (starts with https://script.google.com/...) and paste it in the field above.

That's it! Your data will automatically back up each time you finish a workout. You can also use Push to Sheets manually or Pull from Sheets to restore on a new device.

Tip: If you update the Apps Script code later, you must create a new deployment (Deploy > New deployment) — editing the code alone doesn't update the live URL.
