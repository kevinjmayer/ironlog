Google Sheets Backup Setup

One-time setup (5 min):

1. Open Google Sheets and create a new blank spreadsheet. Name it whatever you want (e.g. "Workout Backup").

2. Go to Extensions > Apps Script.

3. Delete everything in the editor and paste this code:

function doPost(e) { var ss = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.action === 'push') { // --- SAFETY: never let a bad push destroy the backup --- if (!data.workouts || !data.workouts.length) { return ContentService.createTextOutput(JSON.stringify({ ok:false, error:'refused: empty payload' })).setMimeType(ContentService.MimeType.JSON); } var prevRaw = ss.getSheetByName('_raw'); if (prevRaw) { var prevCount = parseInt(prevRaw.getRange(1,4).getValue()) || 0; if (prevCount > 5 && data.workouts.length < prevCount * 0.5 && !data.force) { return ContentService.createTextOutput(JSON.stringify({ ok:false, error:'refused: payload ' + data.workouts.length + ' is under half of stored ' + prevCount })).setMimeType(ContentService.MimeType.JSON); } // Keep one generation of history before overwriting var prevSheet = ss.getSheetByName('_raw_prev'); if (prevSheet) ss.deleteSheet(prevSheet); prevRaw.copyTo(ss).setName('_raw_prev'); // Monthly snapshot: first push of each month archives the PREVIOUS // state, so there's a dated restore point even if recent backups go bad. var now = new Date(); var tag = '_archive_' + now.getFullYear() + '-' + ('0' + (now.getMonth() + 1)).slice(-2); if (!ss.getSheetByName(tag)) { prevRaw.copyTo(ss).setName(tag); } // Prune to the 12 most recent archives. Runs every push so the count // self-corrects rather than only trimming when a new month starts. var archives = ss.getSheets() .map(function(sh) { return sh.getName(); }) .filter(function(n) { return n.indexOf('_archive_') === 0; }) .sort(); while (archives.length > 12) { var oldest = archives.shift(); var sh = ss.getSheetByName(oldest); if (sh) ss.deleteSheet(sh); } } // --- Readable sheet: one row per exercise --- var vis = ss.getSheetByName('Workouts'); if (!vis) vis = ss.insertSheet('Workouts'); vis.clear(); vis.appendRow(['Date','Day','Duration (min)','Readiness','Effort', 'Exercise','Type','Weight (lbs)', 'Set 1','Set 2','Set 3','Set 4','PR','Niggle','Exercise Note','Session Notes']); // Bold + freeze header vis.getRange(1,1,1,16).setFontWeight('bold'); vis.setFrozenRows(1); var rows = []; data.workouts.forEach(function(w) { var prs = w.prs || []; w.exercises.forEach(function(ex) { var r = [w.date, w.dayLabel||'', w.duration||'', w.readiness||'', w.effort||'', ex.name, ex.type||'', ex.weight||'']; var sets = ex.sets || []; for (var i = 0; i < 4; i++) { if (!sets[i]) { r.push(''); continue; } var hasSetWeight = sets[i].weight !== undefined && sets[i].weight !== ex.weight; r.push(hasSetWeight ? sets[i].reps + '@' + sets[i].weight : sets[i].reps); } r.push(prs.indexOf(ex.name) >= 0 ? 'PR' : ''); var lvls = {1:'Tight',2:'Sore',3:'Painful'}; r.push(ex.niggle ? ex.niggle.area + ' (' + (lvls[ex.niggle.level]||'') + ')' : ''); r.push(ex.note || ''); r.push(w.notes || ''); rows.push(r); }); }); if (rows.length > 0) vis.getRange(2,1,rows.length,16).setValues(rows); // --- Sessions sheet: one row per workout --- var ses = ss.getSheetByName('Sessions'); if (!ses) ses = ss.insertSheet('Sessions'); ses.clear(); ses.appendRow(['Date','Day','Duration (min)','Readiness (1-5)','Effort (1-5)', 'Exercises','Total Volume (lbs)','PRs Set','Niggles','Session Notes', 'Coach Adjustments','AI Analysis']); ses.getRange(1,1,1,12).setFontWeight('bold'); ses.setFrozenRows(1); var sRows = []; data.workouts.forEach(function(w) { var vol = 0; (w.exercises||[]).forEach(function(ex) { (ex.sets||[]).forEach(function(s) { var wt = (s.weight !== undefined) ? s.weight : (ex.weight||0); vol += wt * (s.reps||0); }); }); var intent = ''; if (w.noteIntent) { intent = Object.keys(w.noteIntent).map(function(k) { return k + ': ' + w.noteIntent[k].action + (w.noteIntent[k].reason ? ' (' + w.noteIntent[k].reason + ')' : ''); }).join('; '); } var lvls2 = {1:'Tight',2:'Sore',3:'Painful'}; var nigs = (w.exercises||[]).filter(function(ex){ return ex.niggle; }) .map(function(ex){ return ex.niggle.area + ' (' + (lvls2[ex.niggle.level]||'') + ') on ' + ex.name; }).join('; '); sRows.push([w.date, w.dayLabel||'', w.duration||'', w.readiness||'', w.effort||'', (w.exercises||[]).length, Math.round(vol), (w.prs||[]).join(', '), nigs, w.notes||'', intent, w.aiAnalysis||'']); }); if (sRows.length > 0) { ses.getRange(2,1,sRows.length,12).setValues(sRows); ses.setColumnWidth(9, 220); // Niggles ses.setColumnWidth(10, 300); // Session Notes ses.setColumnWidth(12, 400); // AI Analysis } // --- Raw sheet: JSON for reliable restore --- var raw = ss.getSheetByName('_raw'); if (!raw) { raw = ss.insertSheet('_raw'); } raw.clear(); var json = JSON.stringify(data.workouts); var CHUNK = 40000; for (var i = 0; i < json.length; i += CHUNK) raw.getRange(i/CHUNK+1,1).setValue(json.substring(i,i+CHUNK)); raw.getRange(1,2).setValue(new Date().toISOString()); raw.getRange(1,3).setValue(Math.ceil(json.length/CHUNK)); raw.getRange(1,4).setValue(data.workouts.length); return ContentService.createTextOutput(JSON.stringify({ ok:true, count:data.workouts.length, time:new Date().toISOString() })).setMimeType(ContentService.MimeType.JSON); } if (data.action === 'pull') { var raw = ss.getSheetByName('_raw'); if (!raw) return ContentService.createTextOutput( JSON.stringify({ok:true, workouts:[], time:null}) ).setMimeType(ContentService.MimeType.JSON); var n = parseInt(raw.getRange(1,3).getValue()) || 1; var json = ''; for (var i = 0; i < n; i++) json += (raw.getRange(i+1,1).getValue() || ''); var time = raw.getRange(1,2).getValue(); return ContentService.createTextOutput(JSON.stringify({ ok:true, workouts: json ? JSON.parse(json) : [], time: time||null })).setMimeType(ContentService.MimeType.JSON); } return ContentService.createTextOutput( JSON.stringify({ok:false}) ).setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var cb = (e.parameter || {}).callback || ''; var action = (e.parameter || {}).action || 'ping'; var result; if (action === 'pull') { var ss = SpreadsheetApp.getActiveSpreadsheet(); var raw = ss.getSheetByName('_raw'); if (!raw) { result = {ok:true, workouts:[], time:null}; } else { var n = parseInt(raw.getRange(1,3).getValue()) || 1; var json = ''; for (var i = 0; i < n; i++) json += (raw.getRange(i+1,1).getValue() || ''); var time = raw.getRange(1,2).getValue(); result = {ok:true, workouts: json ? JSON.parse(json) : [], time: time||null}; } } else { result = {ok:true, msg:'IronLog Sync active'}; } // JSONP: wrap in callback if provided if (cb) { return ContentService.createTextOutput( cb + '(' + JSON.stringify(result) + ')' ).setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput( JSON.stringify(result) ).setMimeType(ContentService.MimeType.JSON); }

4. Click the 💾 Save icon (or Ctrl+S), then click Deploy > New deployment.

5. Click the ⚙️ gear next to "Select type" and choose Web app.

6. Set "Execute as" to Me. Set "Who has access" to Anyone. Click Deploy.

7. Click Authorize access when prompted. If you see "Google hasn't verified this app", click Advanced > Go to Untitled project (unsafe) — this is safe, it's your own script.

8. Copy the Web app URL (starts with https://script.google.com/...) and paste it in the field above.

That's it! Your data backs up automatically when you finish a workout, and again whenever you edit or delete a past one.

You'll get three sheets: Workouts (one row per exercise), Sessions (one row per workout, with readiness, effort, session notes and the coach's analysis), and _raw (JSON used for restore — leave it alone). A _raw_prev sheet keeps the previous backup as a safety net.

Tip: If you update the Apps Script code later, you must create a new deployment (Deploy > New deployment) — editing the code alone doesn't update the live URL.
